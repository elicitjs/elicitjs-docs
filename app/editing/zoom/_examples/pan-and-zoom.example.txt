// A year of weekly rainfall. Zoom in, then drag the plot to pan along it.
const data = d3.range(53).map((week) => ({
  week,
  rain: Math.round(40 + 25 * Math.sin(week / 8) + 10 * Math.cos(week / 3)),
}));

const chart = Elicit({
  width: 460, height: 260,
  margins: { top: 16, right: 18, bottom: 32, left: 40 },
  schema: {
    week: { type: "quantitative", domain: [0, 52] },
    rain: { type: "quantitative", domain: [0, 100] },
  },
  data,
  // Zoom and pan both write the week domain, so each picks up where the
  // other left off.
  axes: {
    x: { edits: [edit.axis.zoom({ minSpan: 4 }), edit.axis.pan()] },
  },
  marks: [
    lineY({ stroke: "#0284c7", strokeWidth: 2, channels: { x: { field: "week" }, y: { field: "rain" } } }),
    point({
      fill: "#0284c7", size: 4,
      channels: { x: { field: "week" }, y: { field: "rain", edit: move() } },
    }),
  ],
});
mount(chart);

const [LO, HI] = [0, 52];   // the declared week domain
const zoomBy = (k) => {
  const [lo, hi] = chart.getSchema().week.domain;
  const half = Math.max(2, Math.min((hi - lo) * k, HI - LO) / 2);
  const mid = Math.min(Math.max((lo + hi) / 2, LO + half), HI - half);
  chart.setDomain("week", [mid - half, mid + half]);
};

const controls = document.createElement("div");
controls.style.cssText = "display:flex;gap:8px;align-items:center;margin-top:10px;font:12px system-ui";
const addButton = (label, onclick) => {
  const b = document.createElement("button");
  b.textContent = label;
  b.onclick = onclick;
  b.style.cssText = "padding:4px 10px;border:1px solid #cbd5e1;border-radius:6px;background:#fff;cursor:pointer";
  controls.append(b);
};
addButton("Zoom in", () => zoomBy(0.5));
addButton("Zoom out", () => zoomBy(2));
addButton("Reset", () => chart.setDomain("week", null));
const readout = document.createElement("span");
readout.style.color = "#64748b";
controls.append(readout);
mount(controls);

const show = () => {
  const [lo, hi] = chart.getSchema().week.domain;
  readout.textContent = `weeks ${lo.toFixed(1)} – ${hi.toFixed(1)}`;
};
chart.on("change", show);
show();
