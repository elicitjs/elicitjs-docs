const chart = Elicit({
  width: 420, height: 280,
  margins: { top: 16, right: 18, bottom: 32, left: 40 },
  schema: {
    x: { type: "quantitative", domain: [0, 100] },
    y: { type: "quantitative", domain: [0, 100] },
  },
  data: [
    { x: 12, y: 20 }, { x: 24, y: 42 }, { x: 38, y: 35 }, { x: 47, y: 58 },
    { x: 55, y: 51 }, { x: 63, y: 70 }, { x: 71, y: 64 }, { x: 86, y: 82 },
  ],
  // The wheel zooms each axis about the value under the pointer.
  axes: {
    x: { edit: edit.axis.zoom() },
    y: { edit: edit.axis.zoom() },
  },
  marks: [
    point({
      fill: "#8A78C0", size: 6,
      channels: { x: { field: "x" }, y: { field: "y" } },
      edits: [move({ channels: ["x", "y"] })],
    }),
  ],
});
mount(chart);

// Buttons zoom both axes about the centre of the view, clamped to the
// declared domains.
const DECLARED = { x: [0, 100], y: [0, 100] };
const zoomBy = (k) => {
  const schema = chart.getSchema();
  for (const field of ["x", "y"]) {
    const [lo, hi] = schema[field].domain;
    const [LO, HI] = DECLARED[field];
    const half = Math.min((hi - lo) * k, HI - LO) / 2;
    const mid = Math.min(Math.max((lo + hi) / 2, LO + half), HI - half);
    chart.setDomain(field, [mid - half, mid + half]);
  }
};

const controls = document.createElement("div");
controls.style.cssText = "display:flex;gap:8px;margin-top:10px;font:12px system-ui";
const addButton = (label, onclick) => {
  const b = document.createElement("button");
  b.textContent = label;
  b.onclick = onclick;
  b.style.cssText = "padding:4px 10px;border:1px solid #BCC9D4;border-radius:6px;background:#fff;cursor:pointer";
  controls.append(b);
};
addButton("Zoom in", () => zoomBy(0.5));
addButton("Zoom out", () => zoomBy(2));
addButton("Reset", () => { chart.setDomain("x", null); chart.setDomain("y", null); });
mount(controls);
