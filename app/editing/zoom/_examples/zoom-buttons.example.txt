// Buttons drive the x domain from outside the chart; the wheel zooms it too.
const chart = Elicit({
  width: 420, height: 240,
  margins: { top: 16, right: 18, bottom: 32, left: 40 },
  schema: {
    x: { type: "quantitative", domain: [0, 100] },
    y: { type: "quantitative", domain: [0, 10] },
  },
  data: [
    { x: 15, y: 3 }, { x: 32, y: 6 }, { x: 48, y: 4 },
    { x: 66, y: 8 }, { x: 84, y: 5 },
  ],
  axes: { x: { edit: edit.axis.zoom() } },
  marks: [
    point({
      fill: "#2563eb", size: 6,
      channels: { x: { field: "x" }, y: { field: "y", edit: move() } },
    }),
  ],
});
mount(chart);

const [LO, HI] = [0, 100];   // the declared domain
const zoomBy = (k) => {
  const [lo, hi] = chart.getSchema().x.domain;
  const mid = (lo + hi) / 2;
  const half = Math.min((hi - lo) * k, HI - LO) / 2;
  const next = [Math.max(LO, mid - half), Math.min(HI, mid + half)];
  chart.setDomain("x", next);
};

const controls = document.createElement("div");
controls.style.cssText = "display:flex;gap:8px;align-items:center;margin-top:10px;font:12px system-ui";
const addButton = (label, onclick) => {
  const b = document.createElement("button");
  b.textContent = label;
  b.onclick = onclick;
  b.style.cssText = "padding:4px 10px;border:1px solid #cbd5e1;border-radius:6px;background:#fff;cursor:pointer";
  controls.append(b);
};
addButton("Zoom in", () => zoomBy(0.5));
addButton("Zoom out", () => zoomBy(2));
addButton("Reset", () => chart.setDomain("x", null));
const readout = document.createElement("span");
readout.style.color = "#64748b";
controls.append(readout);
mount(controls);

const show = () => {
  const [lo, hi] = chart.getSchema().x.domain;
  readout.textContent = `x: ${lo.toFixed(1)} – ${hi.toFixed(1)}`;
};
chart.on("change", show);
show();
