const chart = Elicit({
  width: 380, height: 250,
  margins: { top: 14, right: 14, bottom: 28, left: 34 },
  data: [
    { cat: "A", n: 62 }, { cat: "B", n: 38 },
    { cat: "C", n: 81 }, { cat: "D", n: 24 },
  ],
  schema: {
    cat: { type: "categorical", domain: ["A", "B", "C", "D"] },
    n:   { type: "quantitative", domain: [0, 100] },
  },
  // `others` styles the bars that are in no state while one bar is selected.
  effects: {
    selected: { others: { opacity: 0.25 } },
  },
  marks: [
    bar({
      channels: { x: { field: "cat" }, y: { field: "n" } },
      edits: [ select({ toggle: true }) ],
    }),
  ],
});
mount(chart.dom());
