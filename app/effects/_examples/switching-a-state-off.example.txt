// `false` disables a state. Here hover and selection are off, so only the drag
// reads back.
mount(Elicit({
  width: 380, height: 240,
  margins: { top: 14, right: 14, bottom: 28, left: 34 },
  data: [
    { cat: "A", n: 62 }, { cat: "B", n: 38 },
    { cat: "C", n: 81 }, { cat: "D", n: 24 },
  ],
  schema: {
    cat: { type: "categorical", domain: ["A", "B", "C", "D"] },
    n:   { type: "quantitative", domain: [0, 100] },
  },
  effects: {
    hovered:  false,
    selected: false,
    grabbed:  { filter: "brightness(0.78) saturate(1.4)" },
  },
  marks: [
    bar({ channels: { x: { field: "cat" }, y: { field: "n", edit: move() } } }),
  ],
}))
