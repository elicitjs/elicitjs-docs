mount(Elicit({
  width: 380, height: 260,
  margins: { top: 14, right: 14, bottom: 26, left: 30 },
  data: [
    { x: 22, y: 32, theta: 90 },
    { x: 74, y: 68, theta: 20 },
  ],
  schema: {
    x:     { type: "quantitative", domain: [0, 100] },
    y:     { type: "quantitative", domain: [0, 100] },
    theta: { type: "quantitative", domain: [-180, 180] },
  },
  marks: [
    tickX({
      length: 36,
      stroke: "#1d4ed8",
      strokeWidth: 4,
      channels: {
        x: { field: "x" },
        // A 4px tick is a small target, so pick the nearest one within 12px
        // instead of demanding a hit. The catchment is measured against the tick
        // as DRAWN — along the arm it currently points down, not the vertical it
        // would occupy at 0° — so the left tick (spun flat) is grabbable off
        // either END, nowhere near the upright segment it started as.
        y: {
          field: "y",
          edit: move({
            channels: ["y"],
            pick: "nearest",
            threshold: 12,
            guide: { catchment: true },
          }),
        },
        angle: { field: "theta", scale: { range: [-180, 180] } },
      },
    }),
  ],
}))
