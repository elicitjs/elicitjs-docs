mount(Elicit({
  width: 340, height: 340,
  margins: { top: 12, right: 12, bottom: 12, left: 12 },
  data: [ {
    curve: 0.7, smirk: 0.5,
    aperture: 0.6, squint: 0.2, browH: 0.5, browT: 0.4,
  } ],
  schema: {
    curve:    { type: "quantitative", domain: [0, 1] },
    smirk:    { type: "quantitative", domain: [0, 1] },
    aperture: { type: "quantitative", domain: [0, 1] },
    squint:   { type: "quantitative", domain: [0, 1] },
    browH:    { type: "quantitative", domain: [0, 1] },
    browT:    { type: "quantitative", domain: [0, 1] },
  },
  marks: [
    face({
      channels: {
        // The mouth is one part, so one drag drives both of its params: the
        // vertical component bends it, the horizontal one tilts it into a smirk.
        mouthCurve: { field: "curve", edit: slide({ axis: "y", increase: "up" }) },
        mouthAsym:  { field: "smirk", edit: slide({ axis: "x", increase: "right" }) },
        // Each eye is one ellipse: rx opens it, ry squints it. Horizontal and
        // vertical drag, so one gesture on an eye reads as reshaping it.
        eyeScale:  { field: "aperture", edit: slide({ axis: "x", increase: "right" }) },
        eyeSquint: { field: "squint",   edit: slide({ axis: "y", increase: "down" }) },
        // Each brow is one tick: height is where it SITS — a positional channel in
        // the face's frame, so `move()` inverts straight back through it — and tilt
        // is a magnitude, so it slides.
        browHeight: { field: "browH", edit: move({ channels: ["y"] }) },
        browTilt:   { field: "browT", edit: slide({ axis: "x", increase: "right" }) },
      },
    }),
  ],
}))
