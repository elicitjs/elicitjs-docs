mount(Elicit({
  width: 340, height: 320,
  margins: { top: 12, right: 12, bottom: 12, left: 12 },
  data: [ { px: 0.1, py: 0.1, valence: 0.4, arousal: 0.2 } ],
  schema: {
    px:      { type: "quantitative", domain: [-1, 1] },
    py:      { type: "quantitative", domain: [-1, 1] },
    valence: { type: "quantitative", domain: [-1, 1] },
    arousal: { type: "quantitative", domain: [-1, 1] },
  },
  marks: [
    face({
      size: 62,
      channels: {
        // The head is an ordinary `point`, so it takes an ordinary `move()`. The
        // eyes, brows and mouth are separate features drawn on top of it, so
        // grabbing one of them edits the expression instead — no arbitration.
        x: { field: "px", edit: move({ channels: ["x", "y"] }) },
        y: { field: "py" },
        mouthCurve: { field: "valence", edit: slide({ axis: "y", increase: "up" }) },
        eyeScale:   { field: "arousal", edit: slide({ axis: "x", increase: "right" }) },
      },
    }),
  ],
}))
