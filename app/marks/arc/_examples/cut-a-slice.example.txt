mount(Elicit({

  margins: { top: 20, right: 20, bottom: 20, left: 20 },

  data: [{ category: "housing", share: 100 }],
  schema: {
    category: {
      type: "categorical",

        edit.stack.edge(),
        edit.stack.merge(),
      ],
    }),
  ],
