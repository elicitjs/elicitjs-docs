const chart = Elicit({
  width: 360, height: 300,
  margins: { top: 16, right: 16, bottom: 28, left: 34 },
  schema: {
    x: { type: "quantitative", domain: [-10, 10] },
    y: { type: "quantitative", domain: [-10, 10] },
    intercept: { type: "quantitative" },
    slope:     { type: "quantitative" },
  },
  data: [{ intercept: 0, slope: 0.6 }],
  marks: [
    // A trend draws its line and two handles, but writes nothing until an edit
    // names the column each handle sets. Staging goes on the edit, like any other:
    // the intercept is live in stage 0, the slope in stage 1.
    trend({
      channels: {
        intercept: { field: "intercept", edit: edit.trend.intercept({ stage: 0 }) },
        slope:     { field: "slope",     edit: edit.trend.slope({ stage: 1 }) },
      },
    }),
  ],
});
mount(chart);
const btn = document.createElement("button");
btn.textContent = "Next: set the slope";
btn.style.marginTop = "8px";
btn.onclick = () => { chart.nextStage(); btn.disabled = true; };
mount(btn);
